Mutation checks for the circuit proofs in fv/v1/zkp/popfv, run 2026-11-08 with fv/v1/lean_run.sh.
Backs the mutant claims in fv/v1/README.md (Commit and Chain rows) and fv/v1/docs/05-spike-results.md (Commit and Chain rows).

Method
- First a clean `lake build` of popfv (Build completed successfully, 18206 jobs, no sorryAx).
- Each mutant is a scratch copy of ONE file under ~/mut/<name>/src/Popfv, compiled with plain `lean` (lakefile leanOptions passed as -D),
  against the real build's .olean files, symlinked, minus the mutated module. The repo files are never changed.
- Lean writes no .olean for a file with errors, so Chain.lean can't import a broken Commit or Pair. For those, a second check
  compiles one file: the mutated module's body, then Chain.lean's body, imports merged ("combo").
- Controls: the same paths with no change. control_commit, control_pair and control_chain compile with rc=0 and no sorryAx.
  The commit combo control has one error, at past_capture_window_passes: `partner_silence_accepts` resolves to another
  declaration once the other imports are open in one file. It is in the control too, and no Chain theorem depends on it.
  All Chain axioms in that control are clean.
- A mutant counts as caught when lean returns rc=1 and the named theorems' `#print axioms` show sorryAx.
  That is a failed proof, not a proved refutation.

set -u
# Runs in fv/v1/zkp/popfv. Each mutant is a scratch copy under ~/mut; the repo files are never changed.
M=$HOME/mut; rm -rf "$M"; mkdir -p "$M"
LP=$(lake env printenv LEAN_PATH)
# The leanOptions of lakefile.toml, which plain `lean` does not read.
OPTS="-DmaxSynthPendingDepth=3 -DrelaxedAutoImplicit=false -Dweak.linter.mathlibStandardSet=true -Dpp.unicode.fun=true"
mutate() { # mutate <name> <file> <old> <new>
  mkdir -p "$M/$1/src/Popfv" "$M/$1/olean/Popfv"
  # Every other Popfv module resolves to the real build, by symlink. The mutated module's entries are removed, so nothing writes through a link.
  ln -s "$PWD"/.lake/build/lib/lean/Popfv/* "$M/$1/olean/Popfv/"
  rm -f "$M/$1/olean/Popfv/$(basename "$2" .lean)".*
  python3 - "$2" "$M/$1/src/Popfv/$(basename "$2")" "$3" "$4" <<'PY'
import sys
src, dst, old, new = sys.argv[1:]
s = open(src).read()
n = s.count(old)
assert n == 1, f"{n} matches for {old!r}"
open(dst, "w").write(s.replace(old, new))
PY
}
check() { # check <name> <file> [<downstream file>]
  local d=$M/$1 f=$2 base; base=$(basename "$f" .lean)
  {
    echo "### $1"
    diff "$f" "$d/src/Popfv/$base.lean" | sed 's/^/  diff: /'
    LEAN_PATH="$d/olean:$LP" timeout 550 lean $OPTS --root="$d/src" -o "$d/olean/Popfv/$base.olean" "$d/src/Popfv/$base.lean" > "$d/out.txt" 2>&1
    echo "  $base.lean rc=$?"
    grep -E ': error'  "$d/out.txt" | cut -c1-220 | sed 's/^/  /'
    grep -E 'depends on axioms|does not depend on any axioms' "$d/out.txt" | cut -c1-220 | sed 's/^/  /'
    if [ -n "${3:-}" ]; then
      if [ -f "$d/olean/Popfv/$base.olean" ]; then
        echo "  mutated $base.olean was written; checking $3 against it"
        LEAN_PATH="$d/olean:$LP" timeout 550 lean $OPTS "$3" > "$d/down.txt" 2>&1
        echo "  $(basename "$3") rc=$?"
        grep -E ': error'  "$d/down.txt" | cut -c1-220 | sed 's/^/  /'
        grep -E 'depends on axioms|does not depend on any axioms' "$d/down.txt" | cut -c1-220 | sed 's/^/  /'
      else
        echo "  no olean written for the mutated $base.lean, so $(basename "$3") was not checked against it"
      fi
    fi
  } > "$d/report.txt" 2>&1
}
C=Popfv/Chain.lean
# Controls: unchanged copies through the same path.
mutate control_commit Popfv/Commit.lean '  | 0 => ⟨leaf, fits 8 idx⟩' '  | 0 => ⟨leaf, fits 8 idx⟩'
mutate control_pair Popfv/Pair.lean '  cb = commit_b ∧' '  cb = commit_b ∧'
mutate control_chain $C '  wpost : fits 33 (' '  wpost : fits 33 ('
mutate commit_fits8_true Popfv/Commit.lean '  | 0 => ⟨leaf, fits 8 idx⟩' '  | 0 => ⟨leaf, True⟩'
mutate pair_cb_true Popfv/Pair.lean '  cb = commit_b ∧' '  True ∧'
mutate pair_upper_true Popfv/Pair.lean '  fits 70 (120 * s - 34300 * n - 1)' '  True'
mutate chain_op_ok_deleted $C $'  /-- line 507 -/\n  op_ok : fits 10 (opF w.t w.role_b w.leaf_p)\n' ''
mutate chain_wpost_true $C '  wpost : fits 33 (pPartnerF w.t + 12000 - aPartnerF w.t w.role_b)' '  wpost : True'
mutate chain_partner_true $C '  partner : partnerAccepts w.xp (asU32 (opF w.t w.role_b w.leaf_p)) w.c_ip w.c_qp' '  partner : True'
mutate chain_root_any $C '  open_p : open_leaves perm w.xp w.leaf_p w.chp (beF w.t 177 32)' '  open_p : ∃ rt, open_leaves perm w.xp w.leaf_p w.chp rt'
mutate chain_fits34 $C '  wpost : fits 33 (' '  wpost : fits 34 ('
check control_commit Popfv/Commit.lean $C &
check control_pair Popfv/Pair.lean $C &
check control_chain $C &
check commit_fits8_true Popfv/Commit.lean $C &
check pair_cb_true Popfv/Pair.lean $C &
check pair_upper_true Popfv/Pair.lean $C &
for n in chain_op_ok_deleted chain_wpost_true chain_partner_true chain_root_any chain_fits34; do check $n $C & done
wait
# Lean writes no .olean when a file has errors, so an upstream mutant can't be imported by Chain.lean.
# Instead: one file holding the mutated module's body followed by Chain.lean's body, with both import lists merged.
combo() { # combo <name> <upstream file>
  local d=$M/$1 base; base=$(basename "$2" .lean)
  python3 - "$d/src/Popfv/$base.lean" Popfv/Chain.lean "$d/src/Popfv/Combo.lean" "Popfv.$base" <<'PY'
import sys
up, ch, out, drop = sys.argv[1:]
def split(p):
    imps, body = [], []
    for line in open(p).read().splitlines():
        (imps if line.startswith("import ") else body).append(line)
    return imps, body
iu, bu = split(up); ic, bc = split(ch)
imps = [i for i in dict.fromkeys(iu + ic) if i != "import " + drop]
open(out, "w").write("\n".join(imps + bu + bc) + "\n")
PY
  {
    echo "### $1, mutated $base.lean body + Chain.lean body in one file"
    echo "  imports: $(grep '^import' "$d/src/Popfv/Combo.lean" | tr '\n' ' ')"
    LEAN_PATH="$d/olean:$LP" timeout 550 lean $OPTS --root="$d/src" "$d/src/Popfv/Combo.lean" > "$d/combo.txt" 2>&1
    echo "  rc=$?"
    grep -E ': error'  "$d/combo.txt" | cut -c1-220 | sed 's/^/  /'
    grep -E 'depends on axioms|does not depend on any axioms' "$d/combo.txt" | grep 'Chain' | cut -c1-220 | sed 's/^/  /'
  } > "$d/combo-report.txt" 2>&1
}
for n in control_commit:Commit control_pair:Pair commit_fits8_true:Commit pair_cb_true:Pair pair_upper_true:Pair; do
  combo "${n%%:*}" "Popfv/${n##*:}.lean" &
done
wait
for n in control_commit control_pair control_chain commit_fits8_true pair_cb_true pair_upper_true chain_op_ok_deleted chain_wpost_true chain_partner_true chain_root_any chain_fits34; do
  cat "$M/$n/report.txt"; echo
done
for n in control_commit control_pair commit_fits8_true pair_cb_true pair_upper_true; do cat "$M/$n/combo-report.txt"; echo; done
